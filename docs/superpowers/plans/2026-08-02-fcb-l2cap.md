import Foundation

/// Pumps one L2CAP channel's stream pair. Ported from ReTiCA's review-hardened
/// ChannelIO. All state is confined to `queue`: CFStream events are scheduled
/// onto it and the public methods hop onto it (send/close use queue.sync — they
/// must never be called FROM the pump queue itself). The designed-for-test
/// initializer takes a raw stream pair so bound in-memory pipes can drive it.
final class L2CAPChannelPump: NSObject, StreamDelegate, @unchecked Sendable {
    private let input: InputStream
    private let output: OutputStream
    private let queue: DispatchQueue

    // Queue-confined state.
    private var outBuffer = Data()
    private var finished = false
    private var incomingContinuation: AsyncStream<Data>.Continuation?
    private var incomingTaken = false
    private var preBuffer: [Data] = []   // inbound chunks before a consumer attaches

    init(input: InputStream, output: OutputStream,
         queue: DispatchQueue = DispatchQueue(label: "fcb.l2cap.pump")) {
        self.input = input
        self.output = output
        self.queue = queue
    }

    func open() {
        queue.async {
            self.input.delegate = self
            self.output.delegate = self
            CFReadStreamSetDispatchQueue(self.input as CFReadStream, self.queue)
            CFWriteStreamSetDispatchQueue(self.output as CFWriteStream, self.queue)
            self.input.open()
            self.output.open()
        }
    }

    /// Single-consumer: the first call returns the live stream (pre-attach
    /// chunks replayed in order); later calls return a finished stream.
    func incomingStream() -> AsyncStream<Data> {
        let (stream, continuation) = AsyncStream<Data>.makeStream()
        queue.async {
            guard !self.incomingTaken else { continuation.finish(); return }
            self.incomingTaken = true
            for chunk in self.preBuffer { continuation.yield(chunk) }
            self.preBuffer = []
            if self.finished {
                continuation.finish()
            } else {
                self.incomingContinuation = continuation
            }
        }
        return stream
    }

    /// Throws only when the channel is already closed/dead.
    func send(_ data: Data) throws {
        try queue.sync {
            guard !finished else { throw BluetoothError.operationCancelled }
            outBuffer.append(data)
            drain()
        }
    }

    /// Idempotent silent teardown; the only observable signal is `incoming` finishing.
    func close() {
        queue.sync { finishOnQueue() }
    }

    // MARK: - StreamDelegate (events arrive on `queue`)

    func stream(_ aStream: Stream, handle eventCode: Stream.Event) {
        switch eventCode {
        case .hasBytesAvailable:
            readAvailable()
        case .hasSpaceAvailable:
            drain()
        case .errorOccurred, .endEncountered:
            finishOnQueue()
        default:
            break
        }
    }

    // MARK: - Queue-confined internals

    private func readAvailable() {
        guard !finished else { return }
        var scratch = [UInt8](repeating: 0, count: 4096)
        while input.hasBytesAvailable {
            let count = input.read(&scratch, maxLength: scratch.count)
            if count > 0 {
                let chunk = Data(scratch[0..<count])
                if let continuation = incomingContinuation {
                    continuation.yield(chunk)
                } else {
                    preBuffer.append(chunk)
                }
            } else if count < 0 {
                finishOnQueue()
                return
            } else {
                break
            }
        }
    }

    private func drain() {
        guard !finished else { return }
        while output.hasSpaceAvailable, !outBuffer.isEmpty {
            let written = outBuffer.withUnsafeBytes { raw in
                output.write(raw.bindMemory(to: UInt8.self).baseAddress!, maxLength: outBuffer.count)
            }
            if written <= 0 { break }
            outBuffer.removeFirst(written)
        }
    }

    private func finishOnQueue() {
        guard !finished else { return }
        finished = true
        outBuffer.removeAll()
        input.close()
        output.close()
        CFReadStreamSetDispatchQueue(input as CFReadStream, nil)
        CFWriteStreamSetDispatchQueue(output as CFWriteStream, nil)
        input.delegate = nil
        output.delegate = nil
        incomingContinuation?.finish()
        incomingContinuation = nil
    }
}

import Foundation

/// A live L2CAP connection. Not a model — a capability handle (carries
/// closures), so it is Identifiable but deliberately not Hashable/Equatable.
/// `incoming()` is single-consumer: the first call returns the live stream,
/// later calls return an already-finished stream. `close()` is silent
/// teardown — the only observable signal is `incoming` finishing.
public struct L2CAPConnection: Sendable, Identifiable {
    public let id: UUID
    public let peerID: UUID
    public let psm: UInt16
    public var incoming: @Sendable () -> AsyncStream<Data>
    public var send: @Sendable (Data) async throws -> Void
    public var close: @Sendable () async -> Void

    public init(
        id: UUID,
        peerID: UUID,
        psm: UInt16,
        incoming: @escaping @Sendable () -> AsyncStream<Data>,
        send: @escaping @Sendable (Data) async throws -> Void,
        close: @escaping @Sendable () async -> Void
    ) {
        self.id = id
        self.peerID = peerID
        self.psm = psm
        self.incoming = incoming
        self.send = send
        self.close = close
    }
}

extension L2CAPConnection {
    /// Wraps a pump into a connection handle. Opens the pump.
    static func make(pump: L2CAPChannelPump, peerID: UUID, psm: UInt16) -> L2CAPConnection {
        pump.open()
        return L2CAPConnection(
            id: UUID(),
            peerID: peerID,
            psm: psm,
            incoming: { pump.incomingStream() },
            send: { data in try pump.send(data) },
            close: { pump.close() }
        )
    }
}

    func openL2CAPChannel(peripheralID: UUID, psm: UInt16) async throws -> L2CAPConnection {
        try preflightCheck()
        guard let connection = peripherals[peripheralID] else {
            throw BluetoothError.unknownPeripheral(peripheralID)
        }
        // CoreBluetoothMock cannot simulate L2CAP — its openL2CAPChannel is a
        // fatalError. Fail loudly-but-gracefully instead of crashing the process.
        guard !(connection is CBMPeripheralMock) else {
            throw BluetoothError.l2capNotSupported
        }
        if pendingOpenL2CAP[peripheralID] != nil {
            throw BluetoothError.operationCancelled
        }
        return try await withCheckedThrowingContinuation { (cont: CheckedContinuation<L2CAPConnection, any Error>) in
            pendingOpenL2CAP[peripheralID] = cont
            connection.openL2CAPChannel(CBML2CAPPSM(psm))
        }
    }

    private func handleDidOpenL2CAP(peripheralID: UUID, peerID: UUID, psm: UInt16, input: InputStream, output: OutputStream) {
        guard let cont = pendingOpenL2CAP.removeValue(forKey: peripheralID) else { return }
        let pump = L2CAPChannelPump(input: input, output: output)
        cont.resume(returning: L2CAPConnection.make(pump: pump, peerID: peerID, psm: psm))
    }

    private func handleDidFailOpenL2CAP(peripheralID: UUID, nsError: NSError?) {
        guard let cont = pendingOpenL2CAP.removeValue(forKey: peripheralID) else { return }
        if let nsError {
            cont.resume(throwing: BluetoothError(nsError))
        } else {
            cont.resume(throwing: BluetoothError.operationCancelled)
        }
    }
